import { Card, CardGrid } from '@astrojs/starlight/components';

<CardGrid stagger>
  <Card title="One source of truth" icon="document">
    Each component is a Markdown file: YAML frontmatter for the schema, prose for the guidance. Docs, props tables, prompts, and code are all projections of it.
  </Card>
  <Card title="Tokens, not values" icon="seti:css">
    W3C Design Tokens built with Style Dictionary into CSS variables, JS, and React Native objects. Semantic names like <code>foreground.strong</code> work on every platform.
  </Card>
  <Card title="Accessibility enforced" icon="approve-check">
    WCAG 2.2 requirements are schema constraints. Contrast is checked against the tokens for every theme and variant at build time.
  </Card>
  <Card title="Made for AI tools" icon="rocket">
    Every schema, guidance section, and generation prompt is a build output — ready to serve through an MCP server so any AI client can generate on-system components.
  </Card>
</CardGrid>
